<body>
    <h2>🐔 雞棚指數 - 海南雞評測</h2>
    <div class="container">
        <div class="controls">
            <div class="control-group">
                <label>餐廳名稱：</label>
                <input type="text" id="restaurantName" value="輸入餐廳名稱" oninput="updateChart()">
            </div>
            
            <div id="sliders"></div>

            <button class="btn" onclick="resetData()">重置數據</button>
            <p class="note">調整滑桿後，右鍵點擊圖表即可儲存圖片</p>
        </div>

        <div class="chart-box">
            <canvas id="radarChart"></canvas>
        </div>
    </div>

    <script>
        // 設定 8 個海南雞評測維度
        const dimensions = [
            { id: 'skin', label: '雞皮爽滑', default: 8 },
            { id: 'meat', label: '肉質嫩滑', default: 8 },
            { id: 'fat', label: '脂肪膠質', default: 6 },
            { id: 'flavor', label: '雞味鮮甜', default: 7 },
            { id: 'rice', label: '油飯香氣', default: 9 },
            { id: 'soy', label: '黑豉油', default: 7 },
            { id: 'ginger', label: '薑蓉', default: 8 },
            { id: 'chili', label: '辣醬', default: 7 }
        ];

        // 動態生成滑桿
        const slidersContainer = document.getElementById('sliders');
        dimensions.forEach(dim => {
            const div = document.createElement('div');
            div.className = 'control-group';
            div.innerHTML = `
                <label>${dim.label}</label>
                <input type="range" id="${dim.id}" min="0" max="10" value="${dim.default}" oninput="updateChart()">
                <span id="val-${dim.id}">${dim.default}</span>
            `;
            slidersContainer.appendChild(div);
        });

        // 初始化圖表
        const ctx = document.getElementById('radarChart').getContext('2d');
        let radarChart = new Chart(ctx, {
            type: 'radar',
            data: {
                labels: dimensions.map(d => d.label),
                datasets: [{
                    label: '雞棚指數',
                    data: dimensions.map(d => d.default),
                    backgroundColor: 'rgba(255, 87, 34, 0.4)', // 橙紅色填充
                    borderColor: 'rgba(255, 87, 34, 1)',       // 橙紅色邊框
                    borderWidth: 2,
                    pointBackgroundColor: '#fff',
                    pointBorderColor: '#ff5722'
                }]
            },
            options: {
                responsive: true,
                plugins: {
                    title: {
                        display: true,
                        text: '輸入餐廳名稱',
                        font: { size: 24, family: 'Microsoft JhengHei' },
                        color: '#333',
                        padding: { bottom: 20 }
                    },
                    legend: { display: false }
                },
                scales: {
                    r: {
                        angleLines: { color: '#ccc' },
                        grid: { color: '#ddd' },
                        pointLabels: {
                            font: { size: 14, weight: 'bold', family: 'Microsoft JhengHei' },
                            color: '#333' 
                        },
                        suggestedMin: 0,
                        suggestedMax: 10,
                        ticks: { stepSize: 2, display: false } 
                    }
                }
            }
        });

        // 更新圖表函數
        function updateChart() {
            // 更新數值顯示
            const newData = dimensions.map(dim => {
                const val = document.getElementById(dim.id).value;
                document.getElementById(`val-${dim.id}`).innerText = val;
                return val;
            });

            // 更新標題
            const name = document.getElementById('restaurantName').value;
            radarChart.options.plugins.title.text = name;

            // 更新數據
            radarChart.data.datasets[0].data = newData;
            radarChart.update();
        }

        // 重置函數
        function resetData() {
            dimensions.forEach(dim => {
                document.getElementById(dim.id).value = 5;
                document.getElementById(`val-${dim.id}`).innerText = 5;
            });
            document.getElementById('restaurantName').value = "我的海南雞食評";
            updateChart();
        }
    </script>

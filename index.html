// 定義兩種模式的維度設定 const modeData = { hainan: [ { id: 'h_skin', label: '外皮', default: 8 }, { id: 'h_meat', label: '肉質', default: 8 }, { id: 'h_fat', label: '脂肪', default: 6 }, { id: 'h_flavor', label: '食味', default: 7 }, { id: 'h_rice', label: '油飯', default: 9 }, { id: 'h_soy', label: '醬油', default: 7 }, { id: 'h_ginger', label: '薑蓉', default: 8 }, { id: 'h_chili', label: '辣醬', default: 7 } ], general: [ { id: 'g_skin', label: '外皮', default: 7 }, { id: 'g_fat', label: '脂肪', default: 6 }, { id: 'g_meat', label: '肉質', default: 8 }, { id: 'g_flavor', label: '食味', default: 7 }, { id: 'g_sauce', label: '調味', default: 7 }, { id: 'g_cook', label: '火候', default: 8 }, { id: 'g_look', label: '賣相', default: 8 }, { id: 'g_cp', label: 'CP值', default: 7 } ] }; let currentDimensions = modeData['hainan']; const slidersContainer = document.getElementById('sliders'); let radarChart; // 1. 初始化圖表 function initChart() { const ctx = document.getElementById('radarChart').getContext('2d'); Chart.defaults.backgroundColor = 'transparent'; radarChart = new Chart(ctx, { type: 'radar', data: { labels: currentDimensions.map(d => d.label), datasets: [{ label: '雞棚指數', data: currentDimensions.map(d => d.default), borderWidth: 3, pointBackgroundColor: '#fff', pointRadius: 4, pointHoverRadius: 6 }] }, options: { responsive: true, maintainAspectRatio: true, plugins: { title: { display: true, font: { size: 28, family: 'Microsoft JhengHei', weight: 'bold' }, padding: { top: 0, bottom: 5 } }, subtitle: { display: true, font: { size: 22, family: 'Microsoft JhengHei', weight: 'bold' }, padding: { bottom: 15 } }, legend: { display: false } }, scales: { r: { pointLabels: { font: { size: 17, weight: 'bold', family: 'Microsoft JhengHei' } }, suggestedMin: 0, suggestedMax: 10, ticks: { stepSize: 2, display: false } } } } }); } // 2. 重新生成滑桿 HTML (不再觸發 updateColors，交由外層控制) function renderSliders() { slidersContainer.innerHTML = currentDimensions.map(dim => `
${dim.default}
`).join(''); } // 3. 處理顏色變更 function updateColors(isStandalone = true) { const rColor = document.getElementById('radarColor').value; const tColor = document.getElementById('textColor').value; const ds = radarChart.data.datasets[0]; const opts = radarChart.options; // 更新圖表顏色 ds.borderColor = rColor; ds.backgroundColor = rColor + '73'; ds.pointBorderColor = rColor; opts.plugins.title.color = tColor; opts.plugins.subtitle.color = rColor; opts.scales.r.pointLabels.color = tColor; opts.scales.r.angleLines.color = tColor + '40'; opts.scales.r.grid.color = tColor + '40'; // 更新拉桿及數值顏色 document.querySelectorAll('span.val-display').forEach(el => el.style.color = rColor); document.querySelectorAll('.control-group input[type="range"]').forEach(el => el.style.accentColor = rColor); if (isStandalone) radarChart.update(); } // 4. 計分與更新圖表數據 function updateChart(isStandalone = true) { let sum = 0; radarChart.data.datasets[0].data = currentDimensions.map(dim => { const val = parseInt(document.getElementById(dim.id).value, 10); document.getElementById(`val-${dim.id}`).innerText = val; sum += val; return val; }); // 動態計算滿分 (就算轉 6 角形都唔會計錯) const maxScore = currentDimensions.length * 10; const totalScore = Math.round((sum / maxScore) * 100); radarChart.options.plugins.title.text = document.getElementById('restaurantName').value; radarChart.options.plugins.subtitle.text = `總分：${totalScore} / 100`; if (isStandalone) radarChart.update(); } // 5. 切換模式觸發函數 function switchMode() { currentDimensions = modeData[document.getElementById('modeSelect').value]; radarChart.data.labels = currentDimensions.map(d => d.label); renderSliders(); updateColors(false); updateChart(true); // 只在最後執行一次 radarChart.update() } // 6. 重置數據函數 function resetData() { document.getElementById('restaurantName').value = "店名：大埔XXXX"; document.getElementById('radarColor').value = "#FFD700"; document.getElementById('textColor').value = "#ffffff"; renderSliders(); updateColors(false); updateChart(true); // 只在最後執行一次 radarChart.update() } // 7. 下載圖片函數 function downloadChart() { const link = document.createElement('a'); link.download = '雞棚指數.png'; link.href = radarChart.canvas.toDataURL('image/png', 1.0); link.click(); } // 初始化 window.onload = () => { initChart(); renderSliders(); updateColors(false); updateChart(true); };
